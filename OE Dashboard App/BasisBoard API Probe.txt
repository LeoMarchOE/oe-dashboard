var BB_KEY  = 'PASTE_YOUR_READONLY_KEY';         // e.g. bb_live_...
var BB_BASE = 'https://api.basisboard.com/v1';

function bbGet_(path) {
  var res = UrlFetchApp.fetch(BB_BASE + path, {
    method: 'get',
    headers: { Authorization: 'Bearer ' + BB_KEY },
    muteHttpExceptions: true
  });
  return { code: res.getResponseCode(), body: res.getContentText() };
}

function bbProbe() {
  var paths = ['/reminders', '/users', '/messages/stats', '/stages'];
  var out = [];
  paths.forEach(function (p) {
    var r;
    try { r = bbGet_(p); }
    catch (e) { out.push('===== GET ' + p + ' -> ERROR ' + e + ' ====='); return; }
    var body = r.body || '';
    out.push('===== GET ' + p + '  (HTTP ' + r.code + ') =====');
    out.push(body.length > 1800 ? body.substring(0, 1800) + '  …[truncated]' : body);
  });
  var txt = out.join('\n\n');
  Logger.log(txt);
  return txt;
}
