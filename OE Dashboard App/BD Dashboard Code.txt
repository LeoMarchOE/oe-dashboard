 *
 * Reads the BasisBoard export ("Basis Master") live and focuses on the two
 * leading indicators BD owns: (1) new Opportunities Created per week, and
 * (2) new GC Relationships added. Everything is a subset/derivative of the
 * same export the other dashboards use — no API key required.
 *
 * SETUP: standalone script. Leave WORKBOOK_ID set to the BB API Data workbook.
 * Add an HTML file named "Index", then Deploy > New deployment > Web app.
 */

var DATA_SHEET  = 'Basis Master';

// (must match BDC_WORKBOOK in Collector_BD.gs).

// ---- Basis Master columns (0-based indexes via colIdx) ----
var COL = { created:'C', submitted:'W', deadline:'E', stage:'U', company:'CB', region:'BQ',
            bdm:'AH', projectedCost:'AV', submittedAmt:'CJ', awarded:'Y', lost:'AA' };
// Stages excluded from the 'created' count (not real pursued opportunities)
var EXCLUDE_RE = /no\s*-?\s*bid|dead|declin/;

var TREND_WEEKS = 13;

function doGet() {
  var viewer = (Session.getActiveUser().getEmail() || '').toLowerCase();
  if (ALLOWED.indexOf(viewer) === -1) return accessDenied(viewer);
  var t = HtmlService.createTemplateFromFile('Index');
  t.data = JSON.stringify(getData_());
  return t.evaluate()
    .setTitle('Optiline — Business Development')
    .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL)
    .addMetaTag('viewport', 'width=device-width, initial-scale=1, viewport-fit=cover');
}

function colIdx_(letter){ var n=0; letter=letter.toUpperCase(); for(var i=0;i<letter.length;i++) n=n*26+(letter.charCodeAt(i)-64); return n-1; }
function toDate_(v){ if(v instanceof Date && !isNaN(v)) return v; if(typeof v==='string' && v.trim()){ var d=new Date(v); if(!isNaN(d)) return d; } return null; }
function toMoney_(v){ if(typeof v==='number') return v; var n=parseFloat(String(v).replace(/[$,\s]/g,'')); return isNaN(n)?0:n; }
function weekStart_(d){ var o=new Date(d.getFullYear(),d.getMonth(),d.getDate()); o.setDate(o.getDate()-((o.getDay()+6)%7)); return o; } // Monday
function addDays_(d,n){ var o=new Date(d); o.setDate(o.getDate()+n); return o; }
function fmtMD_(d){ return Utilities.formatDate(d, Session.getScriptTimeZone()||'America/New_York','M/d'); }
function fmtFull_(d){ return Utilities.formatDate(d, Session.getScriptTimeZone()||'America/New_York','M/d/yyyy'); }

function getData_(){
  var ss = SpreadsheetApp.openById(WORKBOOK_ID);
  var sh = ss.getSheetByName(DATA_SHEET);
  var vals = sh.getDataRange().getValues();
  var iC=colIdx_(COL.created), iSub=colIdx_(COL.submitted), iCo=colIdx_(COL.company),
      iReg=colIdx_(COL.region), iBdm=colIdx_(COL.bdm), iVal=colIdx_(COL.submittedAmt), iStage=colIdx_(COL.stage);

  // Find the BD Validation Date column by header name (robust to column moves).
  var hdr0=(vals[0]||[]).map(function(x){return String(x||'').toLowerCase();});
  var iBDV=-1;
  for(var hh=0;hh<hdr0.length;hh++){ if(/bd\s*valid/.test(hdr0[hh]) && /date/.test(hdr0[hh])){ iBDV=hh; break; } }
  if(iBDV<0){ for(var h2=0;h2<hdr0.length;h2++){ if(/bd\s*valid/.test(hdr0[h2])){ iBDV=h2; break; } } }

  var now=new Date(), yr=now.getFullYear(), jan1=new Date(yr,0,1), curWk=weekStart_(now);
  var inYr=function(d){ return d && d>=jan1 && d<=now; };
  var excluded=function(st){ return EXCLUDE_RE.test(String(st||'').toLowerCase()); };

  // Pursued opportunities: has a Bid Created Date AND stage is not no-bid/dead/declined.
  var rows=[];
  for(var r=1;r<vals.length;r++){
    var row=vals[r];
    var created=toDate_(row[iC]); if(!created) continue;
    if(excluded(row[iStage])) continue;
    rows.push({
      created:created,
      submitted:toDate_(row[iSub]),
      company:String(row[iCo]||'').trim(),
      region:String(row[iReg]||'').trim(),
      bdm:String(row[iBdm]||'').trim()||'Unassigned',
      val:toMoney_(row[iVal]),
      bdVal:(iBDV>=0)?toDate_(row[iBDV]):null
    });
  }

  // ---- Opportunities CREATED (count, by created week) ----
  var crWk={};
  rows.forEach(function(o){ var k=weekStart_(o.created).getTime(); crWk[k]=(crWk[k]||0)+1; });
  var createdWk=crWk[curWk.getTime()]||0, createdWkPrev=crWk[addDays_(curWk,-7).getTime()]||0;
  var createdYtd=rows.filter(function(o){return inYr(o.created);}).length;

  // ---- Submitted $ (Submitted $ CJ, bucketed by SUBMITTED week) ----
  var subWkMap={};
  rows.forEach(function(o){ if(!o.submitted) return; var k=weekStart_(o.submitted).getTime(); if(!subWkMap[k]) subWkMap[k]={dol:0,n:0}; subWkMap[k].dol+=o.val; subWkMap[k].n+=1; });
  var weekly=[];
  for(var i=TREND_WEEKS-1;i>=0;i--){ var ws=addDays_(curWk,-7*i); var e=subWkMap[ws.getTime()]||{dol:0,n:0}; weekly.push({label:fmtMD_(ws),dol:e.dol,n:e.n}); }
  var subWk=subWkMap[curWk.getTime()]||{dol:0,n:0}, subWkPrev=subWkMap[addDays_(curWk,-7).getTime()]||{dol:0,n:0};
  var s4=0; for(var j=1;j<=4;j++){ var w=subWkMap[addDays_(curWk,-7*j).getTime()]; if(w) s4+=w.dol; } var subAvg4=s4/4;
  var subYtdRows=rows.filter(function(o){ return inYr(o.submitted); });
  var subYtdDol=subYtdRows.reduce(function(a,o){return a+o.val;},0), subYtdN=subYtdRows.length;

  // ---- By BD Manager (YTD): created count + submitted $ ----
  var bdmMap={};
  rows.forEach(function(o){
    if(!bdmMap[o.bdm]) bdmMap[o.bdm]={name:o.bdm,createdN:0,subDol:0,subWk:0};
    if(inYr(o.created)) bdmMap[o.bdm].createdN++;
    if(inYr(o.submitted)) bdmMap[o.bdm].subDol+=o.val;
    if(o.submitted && weekStart_(o.submitted).getTime()===curWk.getTime()) bdmMap[o.bdm].subWk+=o.val;
  });
  var byBdm=Object.keys(bdmMap).map(function(k){return bdmMap[k];}).sort(function(a,b){return b.subDol-a.subDol;});

  // ---- New GC relationships (first-seen among pursued opps) ----
  var firstSeen={};
  rows.forEach(function(o){ if(!o.company) return; if(!firstSeen[o.company]||o.created<firstSeen[o.company].date) firstSeen[o.company]={date:o.created,region:o.region}; });
  var companies=Object.keys(firstSeen);
  var newGcWkMap={}; companies.forEach(function(c){ var k=weekStart_(firstSeen[c].date).getTime(); newGcWkMap[k]=(newGcWkMap[k]||0)+1; });
  var newGcWeekly=[]; for(var g=TREND_WEEKS-1;g>=0;g--){ var ws2=addDays_(curWk,-7*g); newGcWeekly.push({label:fmtMD_(ws2),n:newGcWkMap[ws2.getTime()]||0}); }
  var newGcWk=newGcWkMap[curWk.getTime()]||0;
  var newGcYtd=companies.filter(function(c){return inYr(firstSeen[c].date);}).length;
  var ns4=0; for(var h=1;h<=4;h++){ ns4+=(newGcWkMap[addDays_(curWk,-7*h).getTime()]||0);} var newGcAvg4=ns4/4;
  var newestGcs=companies.slice().sort(function(a,b){return firstSeen[b].date-firstSeen[a].date;}).slice(0,12).map(function(c){return {name:c,when:fmtFull_(firstSeen[c].date),region:firstSeen[c].region};});
  var activeGc={}; rows.forEach(function(o){ if(o.company && (inYr(o.created)||inYr(o.submitted))) activeGc[o.company]=1; });
  var activeGcYtd=Object.keys(activeGc).length;

  // ---- By Region (YTD): created count + submitted $ ----
  var regMap={};
  rows.forEach(function(o){ var rg=o.region||'\u2014'; if(!regMap[rg]) regMap[rg]={region:rg,createdN:0,subDol:0}; if(inYr(o.created)) regMap[rg].createdN++; if(inYr(o.submitted)) regMap[rg].subDol+=o.val; });
  var byRegion=Object.keys(regMap).map(function(k){return regMap[k];}).sort(function(a,b){return b.subDol-a.subDol;}).slice(0,12);

  // ---- Top GCs by submitted $ (YTD) ----
  var gcMap={};
  subYtdRows.forEach(function(o){ if(!o.company) return; if(!gcMap[o.company]) gcMap[o.company]={name:o.company,n:0,dol:0}; gcMap[o.company].n++; gcMap[o.company].dol+=o.val; });
  var topGcs=Object.keys(gcMap).map(function(k){return gcMap[k];}).sort(function(a,b){return b.dol-a.dol;}).slice(0,8);

  // ---- BD Validation Date (speed + coverage of pursued opps) ----
  var bdVal=null;
  if(iBDV>=0){
    var createdYtdRows=rows.filter(function(o){ return inYr(o.created); });
    var validatedOfCreated=0, speeds=[], monthly=new Array(12);
    for(var mi=0;mi<12;mi++) monthly[mi]=0;
    createdYtdRows.forEach(function(o){
      if(!o.bdVal) return;
      validatedOfCreated++;
      var days=(o.bdVal-o.created)/86400000;
      if(days>=0 && days<=365) speeds.push(days);
    });
    // validations that OCCURRED this year (by validation date), for the monthly trend
    var validatedYtd=0;
    rows.forEach(function(o){ if(o.bdVal && inYr(o.bdVal)){ validatedYtd++; monthly[o.bdVal.getMonth()]++; } });
    speeds.sort(function(a,b){return a-b;});
    var med=speeds.length? (speeds.length%2? speeds[(speeds.length-1)/2] : (speeds[speeds.length/2-1]+speeds[speeds.length/2])/2) : null;
    bdVal={
      createdYtd:createdYtdRows.length,
      validatedOfCreated:validatedOfCreated,
      coveragePct:createdYtdRows.length? validatedOfCreated/createdYtdRows.length : 0,
      medianDays: med==null? null : Math.round(med),
      validatedYtd:validatedYtd,
      monthly:monthly
    };
  }

  return {
    asOf:fmtFull_(now), weekLabel:fmtMD_(curWk),
    kpi:{ createdWk:createdWk, createdWkPrev:createdWkPrev, createdYtd:createdYtd,
          subWkDol:subWk.dol, subWkN:subWk.n, subWkDolPrev:subWkPrev.dol, subAvg4Dol:subAvg4,
          subYtdDol:subYtdDol, subYtdN:subYtdN, avgOpp:(subYtdN?subYtdDol/subYtdN:0),
          newGcWk:newGcWk, newGcAvg4:newGcAvg4, newGcYtd:newGcYtd, activeGcYtd:activeGcYtd },
    weekly:weekly, byBdm:byBdm, newGcWeekly:newGcWeekly, newestGcs:newestGcs, byRegion:byRegion, topGcs:topGcs,
    funnel: funnel_(ss),
    bdVal: bdVal,
    activity: activity_()
  };
}

/* ================================================================
 * FOLLOW-UP ACTIVITY — reads the BD_Reminders sheet that the
 * collector (bbSyncActivity in Collector.gs) refreshes hourly from
 * the BasisBoard API. The dashboard makes NO API call on page load,
 * so it stays fast and needs no external-request permission at serve
 * time. Computes open / overdue / due-next-7-days + a by-owner table.
 * ================================================================ */
function openSs_(id){
  var last;
  for(var i=0;i<3;i++){ try{ return SpreadsheetApp.openById(id); }catch(e){ last=e; Utilities.sleep(1200*(i+1)); } }
  throw last;
}
function activity_(){
  try{
    var actId = ACTIVITY_WORKBOOK_ID || WORKBOOK_ID;
    var ss = openSs_(actId);
    var rSh=ss.getSheetByName('BD_Reminders');
    if(!rSh) return { available:false, error:'no-sheet' };
    var rv=rSh.getDataRange().getValues();
    if(rv.length<2) return { available:true, open:0, overdue:0, dueSoon:0, noDate:0,
      overduePct:0, byType:{}, owners:[], asOf: syncStamp_(ss) };
    var h=rv[0].map(String);
    var ci=function(n){ return h.indexOf(n); };
    var iName=ci('ownerName'), iType=ci('type'), iDue=ci('dueDateAt'), iStatus=ci('status');
    var now=new Date(), soon=new Date(now.getTime()+7*86400000);
    var open=0, overdue=0, dueSoon=0, noDate=0, byType={}, ownerMap={};
    for(var r=1;r<rv.length;r++){
      var row=rv[r];
      if(iStatus>=0 && String(row[iStatus]||'')!=='open') continue;
      open++;
      var t=String(row[iType]||'other'); byType[t]=(byType[t]||0)+1;
      var dr=row[iDue]; var due=(dr instanceof Date && !isNaN(dr))?dr:(dr?new Date(dr):null);
      var hasDue=(due && !isNaN(due));
      if(hasDue){ if(due<now) overdue++; else if(due<=soon) dueSoon++; } else noDate++;
      var oname=String(row[iName]||'Unassigned')||'Unassigned';
      if(!ownerMap[oname]) ownerMap[oname]={name:oname,open:0,overdue:0,dueSoon:0};
      ownerMap[oname].open++;
      if(hasDue && due<now) ownerMap[oname].overdue++;
      else if(hasDue && due<=soon) ownerMap[oname].dueSoon++;
    }
    var owners=Object.keys(ownerMap).map(function(k){return ownerMap[k];})
      .sort(function(a,b){ return (b.overdue-a.overdue)||(b.open-a.open); }).slice(0,15);
    return { available:true, open:open, overdue:overdue, dueSoon:dueSoon, noDate:noDate,
      overduePct: open? overdue/open : 0, byType:byType, owners:owners, asOf: syncStamp_(ss) };
  }catch(e){ return { available:false, error:String(e) }; }
}
function syncStamp_(ss){
  var s=ss.getSheetByName('BD_Sync'); if(!s) return fmtFull_(new Date());
  var v=s.getRange(1,2).getValue();
  return (v instanceof Date)
    ? fmtFull_(v)+' '+Utilities.formatDate(v, Session.getScriptTimeZone()||'America/New_York','h:mm a')
    : fmtFull_(new Date());
}

/* ================================================================
 * BID FUNNEL — matches the Bid Pace Overview exactly.
 * Reads 'ALL DATA' + 'HP DATES', buckets by Bid Deadline, anchored on
 * YESTERDAY. Invites = every row; Board = stage != Declined; Submitted =
 * not-declined AND (submit date present OR stage Submitted/Awarded/Lost);
 * Hot Prospect = submitted bid whose Project ID appears in HP DATES.
 * ================================================================ */
function fToDate_(v){
  if (v instanceof Date){ var d=new Date(v); d.setHours(0,0,0,0); return d; }
  var st=String(v==null?'':v).trim(); if(!st) return null;
  var mm=/^(\d{1,2})\/(\d{1,2})\/(\d{4})$/.exec(st);
  if(mm) return new Date(+mm[3],+mm[1]-1,+mm[2]);
  var g=new Date(st); return isNaN(g)?null:new Date(g.getFullYear(),g.getMonth(),g.getDate());
}
function funnel_(ss){
  var sh = ss.getSheetByName('ALL DATA'); if(!sh) return null;
  var hpMap={}; var hpSh=ss.getSheetByName('HP DATES');
  if(hpSh){ var hv=hpSh.getDataRange().getValues(); for(var r=1;r<hv.length;r++){ if(hv[r][0]!=='' && fToDate_(hv[r][1])) hpMap[String(hv[r][0]).trim()]=1; } }
  var vals=sh.getDataRange().getValues(); if(vals.length<2) return null;
  var hdr=vals[0].map(String);
  var ci=function(n){ return hdr.indexOf(n); };
  var iDl=ci('Bid deadline - Date'), iStage=ci('Stage'), iReason=ci('Stage reason'),
      iSub=ci('Submitted date - Date'), iSubAmt=ci('Submitted $'), iCr=ci('Created date - Date'), iId=ci('Project ID');
  if(iDl<0||iStage<0) return null;

  var today=new Date(); today.setHours(0,0,0,0);
  var yest=new Date(today.getTime()-86400000);
  var fy=today.getFullYear(), pv=fy-1;
  var prevEnd=new Date(pv, yest.getMonth(), yest.getDate());
  var wins=[];
  for(var m=0;m<12;m++){ wins.push({key:'m'+(m+1), s:new Date(fy,m,1), e:new Date(fy,m+1,0)}); }
  wins.push({key:'ytd', s:new Date(fy,0,1), e:yest});
  wins.push({key:'prev', s:new Date(pv,0,1), e:prevEnd, noCap:true});
  wins.forEach(function(w){ if(!w.noCap && w.e.getTime()>yest.getTime()) w.e=yest; w.st=w.s.getTime(); w.et=w.e.getTime()+86399999;
    w.o={inv:0,created:0,board:0,sub:0,subDol:0,hp:0}; });

  for(var rr=1;rr<vals.length;rr++){
    var row=vals[rr];
    var cr=fToDate_(row[iCr]);
    var stage=String(row[iStage]||'').trim();
    var declined=(stage==='Declined');
    if(cr){ var ct=cr.getTime(); for(var a=0;a<wins.length;a++){ var wa=wins[a]; if(ct>=wa.st&&ct<=wa.et) wa.o.created++; } }
    var dl=fToDate_(row[iDl]); if(!dl) continue;
    var t=dl.getTime();
    var won=(stage==='Awarded'), lost=(stage==='Lost');
    var submitted=!declined && ((iSub>=0 && String(row[iSub]||'').trim()!=='') || stage==='Submitted' || won || lost);
    var hp = (iId>=0 && hpMap[String(row[iId]||'').trim()]) ? true : false;
    var subAmt=(iSubAmt>=0)?(Number(row[iSubAmt])||0):0;
    for(var b=0;b<wins.length;b++){
      var w=wins[b]; if(t<w.st||t>w.et) continue;
      w.o.inv++;
      if(declined) continue;
      w.o.board++;
      if(submitted){ w.o.sub++; w.o.subDol+=subAmt; if(hp) w.o.hp++; }
    }
  }
  return {
    anchor: (PD_M_[yest.getMonth()]+' '+yest.getDate()+', '+yest.getFullYear()),
    year: fy, prevYear: pv,
    cols: wins.map(function(w){ return {key:w.key, inv:w.o.inv, created:w.o.created, board:w.o.board, sub:w.o.sub, subDol:w.o.subDol, hp:w.o.hp}; })
  };
}

  var html = '<!DOCTYPE html><html><head><meta name="viewport" content="width=device-width, initial-scale=1">'
    + '<style>body{margin:0;background:#000;color:#fff;font-family:Inter,Arial,sans-serif;min-height:100vh;display:flex;align-items:center;justify-content:center;text-align:center}'
    + '.box{max-width:440px;padding:32px}.bar{width:90px;height:9px;background:#E1251B;margin:0 auto 22px}'
    + 'h1{font-size:22px;letter-spacing:3px;margin:0 0 14px;font-weight:800}p{color:#bbb;font-size:14px;line-height:1.65}b{color:#E1251B}</style></head>'
    + '<body><div class="box"><div class="bar"></div><h1>ACCESS RESTRICTED</h1>'
    + '<p>This dashboard is limited to approved Optiline users.'
    + (email?'<br><br>Signed in as <b>'+email+'</b>':'<br><br>Could not confirm your Google identity.')
    + '<br><br>If you need access, contact the dashboard administrator.</p></div></body></html>';
  return HtmlService.createHtmlOutput(html).setTitle('Access Restricted').setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL);
}
