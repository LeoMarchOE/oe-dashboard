/**
 * Optiline — BD ACTIVITY COLLECTOR
 *
 * Pulls BasisBoard reminders + users from the live API and writes them into
 * dedicated tabs the BD Dashboard reads:  BD_Reminders, BD_Users, BD_Sync.
 * This is the piece that touches the API — run it manually once, then install
 * the daily 6am trigger.  The dashboard itself never calls the API, so it loads
 * fast and needs no external-request permission at serve time.
 *
 * SETUP
 *   1) Put this file in the SAME Apps Script project as your BD Dashboard
 *      (the one where the probe already reached the API).
 *   2) Set the API key ONE of two ways:
 *        a) Project Settings > Script properties > BASIS_API_KEY = bb_live_...
 *        b) or paste it into BDC_KEY below.
 *      (Script property wins if both are present.)
 *   3) Run bbSyncActivity() once  -> approve the auth prompt -> check the log.
 *   4) Run bbInstallActivityTrigger() once to refresh daily at 6am automatically.
 */

var BDC_BASE     = 'https://api.basisboard.com/v1';
// Where the BD_Reminders / BD_Users / BD_Sync tabs get written.
// The big BB API Data workbook can be slow to open (SpreadsheetApp timeouts).
// For speed + reliability, create a small blank Google Sheet and paste its id
// here (and use the SAME id for ACTIVITY_WORKBOOK_ID in the dashboard Code.gs).
var BDC_WORKBOOK = '1f8KrpUTLRi2_pgHTBZwapB8Ar5NEacHvR46gNVFblrw';
var BDC_KEY      = '';  // optional fallback if you don't use the Script property

/** Open a spreadsheet with retries (guards against transient service timeouts). */
function bdcOpen_(id){
  var last;
  for(var i=0;i<4;i++){
    try{ return SpreadsheetApp.openById(id); }
    catch(e){ last=e; Utilities.sleep(1500*(i+1)); }
  }
  throw last;
}

function bdcKey_(){
  var k=(PropertiesService.getScriptProperties().getProperty('BASIS_API_KEY')||'').trim();
  return k || (BDC_KEY||'').trim();
}
function bdcGet_(path){
  var key=bdcKey_();
  if(!key) throw new Error('No API key: set Script property BASIS_API_KEY or fill BDC_KEY.');
  var res=UrlFetchApp.fetch(BDC_BASE+path, {
    method:'get', muteHttpExceptions:true,
    headers:{ 'Authorization':'Bearer '+key, 'Accept':'application/json' }
  });
  var code=res.getResponseCode();
  if(code<200||code>=300) throw new Error('HTTP '+code+' on '+path+' :: '+res.getContentText().slice(0,200));
  return JSON.parse(res.getContentText());
}

/** Main: refresh BD_Users, BD_Reminders (open), BD_Sync. */
function bbSyncActivity(){
  // Fetch from the API FIRST (fast), then open the sheet only to write.
  var users=(bdcGet_('/users?limit=1000').users)||[];
  var uMap={};
  users.forEach(function(u){ uMap[u.id]=((u.firstName||'')+' '+(u.lastName||'')).trim()||u.email||'Unknown'; });

  var rems=(bdcGet_('/reminders?filter[status]=open&limit=1000').reminders)||[];

  // Open the workbook (with retries) only now, to write.
  var ss=bdcOpen_(BDC_WORKBOOK);

  // ---- Users ----
  var uSh=ss.getSheetByName('BD_Users')||ss.insertSheet('BD_Users');
  uSh.clearContents();
  var uRows=[['id','name','email','position','active']];
  users.forEach(function(u){ uRows.push([u.id||'', uMap[u.id], u.email||'', u.position||'', u.active===true]); });
  uSh.getRange(1,1,uRows.length,uRows[0].length).setValues(uRows);

  // ---- Reminders (open) ----
  var rSh=ss.getSheetByName('BD_Reminders')||ss.insertSheet('BD_Reminders');
  rSh.clearContents();
  var rRows=[['id','name','status','type','dueDateAt','ownerId','ownerName','projectName','projectId']];
  rems.forEach(function(rm){
    if(rm.deletedAt) return;
    var oid=(rm.assigneeIds&&rm.assigneeIds[0])||'';
    rRows.push([
      rm.id||'', rm.name||'', rm.status||'', rm.type||'',
      rm.dueDateAt? new Date(rm.dueDateAt): '',
      oid, uMap[oid]||'Unassigned',
      (rm.project&&rm.project.name)||'', (rm.project&&rm.project.id)||''
    ]);
  });
  rSh.getRange(1,1,rRows.length,rRows[0].length).setValues(rRows);

  // ---- Sync stamp ----
  var meta=ss.getSheetByName('BD_Sync')||ss.insertSheet('BD_Sync');
  meta.clearContents();
  meta.getRange(1,1,1,2).setValues([['lastSync', new Date()]]);

  Logger.log('BD sync OK — '+users.length+' users, '+(rRows.length-1)+' open reminders.');
}

/** Run once: refresh daily at 6am (script time zone). */
function bbInstallActivityTrigger(){
  ScriptApp.getProjectTriggers().forEach(function(t){
    if(t.getHandlerFunction()==='bbSyncActivity') ScriptApp.deleteTrigger(t);
  });
  ScriptApp.newTrigger('bbSyncActivity').timeBased().atHour(6).nearMinute(0).everyDays(1).create();
  Logger.log('Daily 6am bbSyncActivity trigger installed.');
}
